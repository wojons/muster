  helm install --namespace openapi-cli openapi-cli ./chart
  helm upgrade --namespace openapi-cli openapi-cli ./chart

{{- if .Values.ingress.enabled }}
{{- $fullName := include "openapi-cli.fullname" . -}}
{{- $svcPort := .Values.service.port -}}
Thank you for installing {{ .Chart.Name }}!

Your release is named {{ .Release.Name }}.

To get started, run the following commands:

{{- range $host := .Values.ingress.hosts }}
  {{$host.host}} has been configured:

  {{- range .paths }}
  http{{ if $.Values.ingress.tls }}s{{ end }}://{{ $host.host }}{{ .path }}
  {{- end }}
{{- end }}

{{- else if contains "NodePort" .Values.service.type }}
  Get the application URL by running these commands:
  export NODE_PORT=$(kubectl get --namespace {{ .Release.Namespace }} -o jsonpath="{.spec.ports[0].nodePort}" services {{ include "openapi-cli.fullname" . }})
  export NODE_IP=$(kubectl get nodes --namespace {{ .Release.Namespace }} -o jsonpath="{.items[0].status.addresses[0].address}")
  echo http://$NODE_IP:$NODE_PORT
{{- else if contains "LoadBalancer" .Values.service.type }}
  Get the application URL by running these commands:
     NOTE: It may take a few minutes for the LoadBalancer IP to be available.
           You can watch the status by running 'kubectl get --namespace {{ .Release.Namespace }} svc -w {{ include "openapi-cli.fullname" . }}'
  export SERVICE_IP=$(kubectl get svc --namespace {{ .Release.Namespace }} {{ include "openapi-cli.fullname" . }} --template "{{"{{ range (index .status.loadBalancer.ingress 0) }}{{.}}{{ end }}"}}")
  echo http://$SERVICE_IP:{{ .Values.service.port }}
{{- else if contains "ClusterIP" .Values.service.type }}
  Get the application URL by running these commands:
  export POD_NAME=$(kubectl get pods --namespace {{ .Release.Namespace }} -l "app.kubernetes.io/name={{ include "openapi-cli.name" . }},app.kubernetes.io/instance={{ .Release.Name }}" -o jsonpath="{.items[0].metadata.name}")
  export CONTAINER_PORT=$(kubectl get pod --namespace {{ .Release.Namespace }} $POD_NAME -o jsonpath="{.spec.containers[{(index .spec.containers 0)}].ports[{(index .spec.containers 0).ports[0]}].containerPort}")
  echo "Visit http://127.0.0.1:8080 to use your application"
  kubectl --namespace {{ .Release.Namespace }} port-forward $POD_NAME 8080:$CONTAINER_PORT
{{- end }}

To check the deployment status:

  kubectl get pods --namespace {{ .Release.Namespace }} -l "app.kubernetes.io/name={{ include "openapi-cli.name" . }}"

To view logs:

  kubectl logs --namespace {{ .Release.Namespace }} -l "app.kubernetes.io/name={{ include "openapi-cli.name" . }}"

To get the service details:

  export SERVICE_PORT=$(kubectl get svc --namespace {{ .Release.Namespace }} {{ include "openapi-cli.fullname" . }} --template "{{"{{ (index .spec.ports 0).port }}"}}")
  echo "Service Port: $SERVICE_PORT"

For more information, see: https://openapi-cli.io/docs/
