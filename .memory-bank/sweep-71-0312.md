type MemoryCache struct {
    mu          sync.RWMutex  // Single mutex for all operations
    data        map[string]*CacheEntry  // Standard Go map
    maxSize     int64
    currentSize int64
    eviction    EvictionPolicy
    stats       Stats
    closed      bool
    statsMu     sync.Mutex
}

// NO bucket-level locks
// NO MapEntry chaining
// NO fine-grained locking
// NO version based optimistic locking
// Just simple sync.RWMutex around map access

func (mc *MemoryCache) Get(ctx context.Context, key string) (Value, error) {
    mc.mu.Lock()  // Single lock (not fine-grained)
    defer mc.mu.Unlock()
    
    if mc.closed {
        return Value{}, ErrClosed
    }
    
    e, exists := mc.data[key]
    if !exists {
        mc.recordMiss()
        return Value{}, ErrNotFound
    }
    
    if e.Value.IsExpired() {
        mc.recordMiss()
        return Value{}, ErrExpired
    }
    
    e.AccessedAt = time.Now()
    e.AccessCount++
    mc.recordHit()
    // ... rest of implementation
}
